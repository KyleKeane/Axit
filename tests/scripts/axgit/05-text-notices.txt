# 5.1 Find in the details after a diff: the match selected and read, F3 the next, "Not found" at the end.
keys enter
expect said Diff of a.txt
expect focus "value":"diff --git
keys ctrl+f
expect state "notice":"Find"
expect focus "name":"Text to find"
wait 1200
type marker
keys enter
expect state "notice":null
expect said → Diff of a.txt (Text): +first paragraph marker
keys f3
expect said · +second paragraph marker
# The bundle's find wraps once (AxitForm.FindNext), so F3 after the last match goes back to the first; "Not found"
# comes only for a text with no match at all (SPEC AG-2.4). The plan's "Not found at the end" is out of date.
keys f3
expect said · +first paragraph marker
keys ctrl+f
expect state "notice":"Find"
wait 1200
keys ctrl+a
type absent
keys enter
expect said ! Not found
# Ctrl+Home, Page Down: a screen down (how far depends on the window's height on the test desktop).

expect said · diff --git

expect said ·
# Ctrl+Down: the next paragraph. A diff has no blank lines (a blank context line is one space), so a commit's

keys ctrl+2

expect said lines
expect focus "value":"commit

keys ctrl+down
