# 1.1: README.md: the title, the text named after the file with its first line; nothing else spoken.
start preview {temp}\README.md

check commands
expect state "title":"README.md - axPreview"

# Axit, Help, and no Edit menu (the runner has no refuse for the commands, so the absence is not checked).

expect commands "Word wrap, Options, Text, Ctrl\u002BShift\u002BW, checked"
expect commands "Keyboard shortcuts, Help, F1"
